2. Dual Review Test Suite Implementation (xUnit)

[Fact]
public async Task SubmitDualReview_WhenValidAndCompleted_ReturnsOk()
{
    // Arrange
    var bookingId = "SB-RAMNAD-01";
    var request = new SubmitReviewRequest
    {
        BookingId = bookingId,
        BranchId = "b1",
        BranchRating = 5,
        BranchComment = "Clean and prompt service in Ramnad Town",
        StylistId = "s1",
        StylistRating = 5,
        StylistComment = "Karthik did a fantastic haircut",
        CustomerName = "Mohamed Irfan"
    };

    // Act
    var result = await _bookingController.SubmitReview(request);

    // Assert
    var okResult = Assert.IsType<OkObjectResult>(result);
    Assert.NotNull(okResult.Value);
}

[Fact]
public async Task SubmitDualReview_WhenDuplicate_ReturnsBadRequest()
{
    // Arrange: same booking reviewed twice
    var bookingId = "SB-RAMNAD-01";
    var request = new SubmitReviewRequest { BookingId = bookingId, BranchRating = 5, StylistRating = 5 };

    // Act
    var firstAttempt = await _bookingController.SubmitReview(request);
    var secondAttempt = await _bookingController.SubmitReview(request);

    // Assert
    Assert.IsType<BadRequestObjectResult>(secondAttempt);
}
