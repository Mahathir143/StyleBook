2. Sample xUnit Financial Test Suite (C#)

using Xunit;
using StyleBook.Shared.Domain.Entities;

namespace StyleBook.Tests.Financial
{
    public class CommissionAndAccountingTests
    {
        [Theory]
        [InlineData(1000.00, 10.00, 100.00, 18.00, 882.00)]
        [InlineData(2500.00, 8.00, 200.00, 36.00, 2264.00)]
        [InlineData(800.00, 12.00, 96.00, 17.28, 686.72)]
        public void Test_BookingCommission_Calculations(
            decimal gross, decimal commissionPercent, 
            decimal expectedCommission, decimal expectedGst, decimal expectedOwnerPayable)
        {
            // Act
            decimal commission = gross * (commissionPercent / 100m);
            decimal gst = commission * 0.18m;
            decimal ownerNet = gross - commission - gst;

            // Assert
            Assert.Equal(expectedCommission, commission);
            Assert.Equal(expectedGst, gst);
            Assert.Equal(expectedOwnerPayable, ownerNet);
        }

        [Fact]
        public void Test_OwnerPrepaidWallet_Blocks_WhenNegativeLimitExceeded()
        {
            // Arrange
            var wallet = new OwnerWallet
            {
                CurrentBalance = -1950.00m,
                NegativeCreditLimit = -2000.00m,
                IsPayoutBlocked = false
            };

            decimal cashTripCommissionToDebit = 118.00m; // ₹100 + ₹18 GST

            // Act
            wallet.CurrentBalance -= cashTripCommissionToDebit; // Becomes -2068.00
            if (wallet.CurrentBalance < wallet.NegativeCreditLimit)
            {
                wallet.IsPayoutBlocked = true;
            }

            // Assert
            Assert.True(wallet.IsPayoutBlocked, "Wallet must lock booking acceptance when balance breaches credit limit.");
        }
    }
}
